// กำหนด ID ของ Google Sheets เพื่อเป็น Database / Backup const SHEET_ID = '1oUEnoXoVheza0RlbRpTrO57h6a0-MCqSX__u8FI6mbg'; function setup() { SpreadsheetApp.openById(SHEET_ID); Logger.log("✅ ยืนยันสิทธิ์สำเร็จเรียบร้อย"); } function sendResponse(response) { return ContentService.createTextOutput(JSON.stringify(response)) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { try { // แยกระบบชัดเจน หากเรียกจาก GitHub จะดึงข้อมูล JSON ให้ if (e.parameter.action === 'getData') { var monthYear = e.parameter.monthYear; var fullName = e.parameter.fullName; var ss = SpreadsheetApp.openById(SHEET_ID); var dbSheet = ss.getSheetByName("WorkReports"); if (!dbSheet) return sendResponse({ status: 'success', tasks: [] }); var values = dbSheet.getDataRange().getValues(); var tasks = []; for (var i = 1; i < values.length; i++) { if (values[i][1] === monthYear && values[i][2] === fullName) { var jsonStr = values[i][5]; if(jsonStr) { try { tasks = JSON.parse(jsonStr); } catch(err) {} } break; } } return sendResponse({ status: 'success', tasks: tasks }); } // หากเข้าหน้าเว็บผ่าน URL ของ Apps Script ตรงๆ จะแสดงหน้าแอปให้ return HtmlService.createHtmlOutputFromFile('Index') .setTitle('🌸 Smart Work Report') .addMetaTag('viewport', 'width=device-width, initial-scale=1'); } catch (error) { return sendResponse({ status: 'error', message: error.message }); } } function doPost(e) { try { var data; // เลี่ยง CORS Block โดยการรับข้อมูลแบบ URL Encoded if (e.parameter.data) { data = JSON.parse(e.parameter.data); } else if (e.postData && e.postData.contents) { data = JSON.parse(e.postData.contents); } else { return sendResponse({ status: 'error', message: 'No data' }); } var ss = SpreadsheetApp.openById(SHEET_ID); var dbSheet = ss.getSheetByName("WorkReports"); if (!dbSheet) { dbSheet = ss.insertSheet("WorkReports"); dbSheet.appendRow(["Timestamp", "เดือน/ปี", "ชื่อ-สกุล", "ตำแหน่ง", "ข้อมูลอ่านง่าย (ให้อ่าน)", "ข้อมูลระบบ (JSON)"]); dbSheet.getRange("A1:F1").setFontWeight("bold").setBackground("#d9ead3"); } var values = dbSheet.getDataRange().getValues(); var isUpdated = false; var prettyTasks = data.tasks.map(function(t) { return t.category + ": \n" + t.detail; }).join("\n\n"); var rawJson = JSON.stringify(data.tasks); for (var i = 1; i < values.length; i++) { if (values[i][1] === data.monthYear && values[i][2] === data.fullName) { dbSheet.getRange(i + 1, 1).setValue(new Date()); dbSheet.getRange(i + 1, 4).setValue(data.position); dbSheet.getRange(i + 1, 5).setValue(prettyTasks); dbSheet.getRange(i + 1, 6).setValue(rawJson); isUpdated = true; break; } } if (!isUpdated) { dbSheet.appendRow([new Date(), data.monthYear, data.fullName, data.position, prettyTasks, rawJson]); } return sendResponse({ status: 'success' }); } catch (error) { return sendResponse({ status: 'error', message: error.message }); } }
